10) Inventory Reorder Checker

import java.util.*;
import java.util.Scanner;
public class inventoryreorderchecker
{
    public static void main(String[] args)
    {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Enter the current stock quantity: ");
        int stockQuantity = scanner.nextInt();
        System.out.println("Enter the minimum stock level: ");
        int minStockLevel = scanner.nextInt();
        System.out.println("Enter the incoming stock quantity: ");
        int incomingStock = scanner.nextInt();

        int currentStock = stockQuantity + incomingStock; // Example reorder level calculation

        String reorderStatus = (currentStock <= minStockLevel) ? "Reorder Required" : "Stock Sufficient";
        System.out.println("Reorder Status: " + reorderStatus);
    }    
}
